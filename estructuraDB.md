# Estructura de Base de Datos — Arteconfi (PostgreSQL)



```sql
-- ============================================================================
-- ARTECONFI - Esquema PostgreSQL
-- Compatible con PostgreSQL 15+
-- ============================================================================

BEGIN;

-- ----------------------------------------------------------------------------
-- 1. MÓDULO USUARIOS Y CONFIGURACIÓN
-- ----------------------------------------------------------------------------

CREATE TABLE usuarios_user (
    id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    password        VARCHAR(128) NOT NULL,
    last_login      TIMESTAMPTZ NULL,
    is_superuser    BOOLEAN NOT NULL DEFAULT FALSE,
    username        VARCHAR(150) NOT NULL UNIQUE,
    first_name      VARCHAR(150) NOT NULL DEFAULT '',
    last_name       VARCHAR(150) NOT NULL DEFAULT '',
    email           VARCHAR(254) NOT NULL UNIQUE,
    is_staff        BOOLEAN NOT NULL DEFAULT FALSE,
    is_active       BOOLEAN NOT NULL DEFAULT TRUE,
    date_joined     TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Token de autenticación para Dueño / Personal (Django REST Framework)
CREATE TABLE authtoken_token (
    key         VARCHAR(40) PRIMARY KEY,
    created     TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    user_id     BIGINT NOT NULL UNIQUE REFERENCES usuarios_user(id) ON DELETE CASCADE
);

CREATE TABLE configuracion_negocio (
    id                   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    amber_days           INTEGER NOT NULL DEFAULT 30 CHECK (amber_days >= 0),
    red_days             INTEGER NOT NULL DEFAULT 15 CHECK (red_days >= 0),
    inactivity_minutes   INTEGER NOT NULL DEFAULT 15 CHECK (inactivity_minutes >= 0),
    codigo_recuperacion  VARCHAR(32) NOT NULL DEFAULT 'ARTE-9482-CONF',
    failed_attempts      INTEGER NOT NULL DEFAULT 0 CHECK (failed_attempts >= 0),
    locked_until         TIMESTAMPTZ NULL
);

-- ----------------------------------------------------------------------------
-- 2. MÓDULO INVENTARIO, LOTES, COMBOS Y KARDEX
-- ----------------------------------------------------------------------------

CREATE TABLE categorias (
    id_categoria      INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre_categoria  VARCHAR(50) NOT NULL UNIQUE,
    descripcion       TEXT NULL
);

CREATE TABLE proveedores (
    id_proveedor     INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre_empresa   VARCHAR(100) NOT NULL,
    rut              VARCHAR(20) NULL,
    contacto_nombre  VARCHAR(100) NULL,
    telefono         VARCHAR(20) NULL,
    email            VARCHAR(100) NULL,
    dias_entrega     VARCHAR(80) NULL,
    direccion        VARCHAR(150) NULL,
    activo           BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE productos (
    id_producto        INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    codigo_barras      VARCHAR(50) NULL UNIQUE,
    nombre             VARCHAR(100) NOT NULL,
    descripcion        TEXT NULL,
    id_categoria       INTEGER NOT NULL REFERENCES categorias(id_categoria) ON DELETE RESTRICT,
    id_proveedor       INTEGER NULL REFERENCES proveedores(id_proveedor) ON DELETE SET NULL,
    precio_compra      NUMERIC(10, 2) NOT NULL CHECK (precio_compra >= 0),
    precio_venta       NUMERIC(10, 2) NOT NULL CHECK (precio_venta >= 0),
    stock_actual       INTEGER NOT NULL DEFAULT 0 CHECK (stock_actual >= 0),
    stock_minimo       INTEGER NOT NULL DEFAULT 5 CHECK (stock_minimo >= 0),
    stock_objetivo     INTEGER NOT NULL DEFAULT 20 CHECK (stock_objetivo >= 0),
    unidad_medida      VARCHAR(10) NOT NULL DEFAULT 'Unidad'
                       CHECK (unidad_medida IN ('Unidad', 'Gramos', 'Kilos')),
    fecha_vencimiento  DATE NULL,
    perecible          BOOLEAN NOT NULL DEFAULT TRUE,
    activo             BOOLEAN NOT NULL DEFAULT TRUE,
    imagen             VARCHAR(100) NULL
);

CREATE TABLE productos_proveedores (
    id            BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    producto_id   INTEGER NOT NULL REFERENCES productos(id_producto) ON DELETE CASCADE,
    proveedor_id  INTEGER NOT NULL REFERENCES proveedores(id_proveedor) ON DELETE CASCADE,
    UNIQUE (producto_id, proveedor_id)
);

CREATE TABLE historial_precios (
    id             BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    producto_id    INTEGER NOT NULL REFERENCES productos(id_producto) ON DELETE CASCADE,
    fecha          DATE NOT NULL DEFAULT CURRENT_DATE,
    precio_compra  NUMERIC(10, 2) NOT NULL CHECK (precio_compra >= 0),
    precio_venta   NUMERIC(10, 2) NOT NULL CHECK (precio_venta >= 0),
    motivo         VARCHAR(150) NOT NULL DEFAULT 'Actualización manual'
);

CREATE TABLE lotes (
    id                 BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    producto_id        INTEGER NOT NULL REFERENCES productos(id_producto) ON DELETE CASCADE,
    codigo_lote        VARCHAR(50) NOT NULL,
    fecha_vencimiento  DATE NULL,
    precio_compra      NUMERIC(10, 2) NOT NULL CHECK (precio_compra >= 0),
    stock_actual       INTEGER NOT NULL DEFAULT 0 CHECK (stock_actual >= 0),
    precio_oferta      NUMERIC(10, 2) NULL CHECK (precio_oferta IS NULL OR precio_oferta >= 0)
);

CREATE TABLE combos (
    id               BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre           VARCHAR(100) NOT NULL,
    precio_especial  NUMERIC(10, 2) NOT NULL CHECK (precio_especial >= 0),
    fecha_inicio     DATE NULL,
    fecha_fin        DATE NULL,
    activo           BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE combo_items (
    id           BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    combo_id     BIGINT NOT NULL REFERENCES combos(id) ON DELETE CASCADE,
    producto_id  INTEGER NOT NULL REFERENCES productos(id_producto) ON DELETE CASCADE,
    cantidad     INTEGER NOT NULL DEFAULT 1 CHECK (cantidad > 0)
);

CREATE TABLE movimientos_inventario (
    id                BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    fecha_hora        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    tipo              VARCHAR(30) NOT NULL
                      CHECK (tipo IN (
                          'Ingreso',
                          'Venta',
                          'Merma',
                          'Reserva Encargo',
                          'Devolución Cliente',
                          'Devolución Proveedor'
                      )),
    producto_id       INTEGER NOT NULL REFERENCES productos(id_producto) ON DELETE RESTRICT,
    lote_id           BIGINT NULL REFERENCES lotes(id) ON DELETE SET NULL,
    codigo_lote       VARCHAR(50) NOT NULL DEFAULT '-',
    cantidad          INTEGER NOT NULL,
    stock_resultante  INTEGER NOT NULL CHECK (stock_resultante >= 0),
    costo_unitario    NUMERIC(10, 2) NOT NULL DEFAULT 0 CHECK (costo_unitario >= 0),
    motivo            VARCHAR(255) NOT NULL
);

-- ----------------------------------------------------------------------------
-- 3. MÓDULO COMPRAS Y PROVEEDORES
-- ----------------------------------------------------------------------------

CREATE TABLE ordenes_compra (
    id            BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    fecha         DATE NOT NULL DEFAULT CURRENT_DATE,
    proveedor_id  INTEGER NOT NULL REFERENCES proveedores(id_proveedor) ON DELETE RESTRICT,
    estado        VARCHAR(20) NOT NULL DEFAULT 'Borrador'
                  CHECK (estado IN ('Borrador', 'Enviada', 'Recibida', 'Cancelada'))
);

CREATE TABLE detalle_ordenes_compra (
    id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    orden_id        BIGINT NOT NULL REFERENCES ordenes_compra(id) ON DELETE CASCADE,
    producto_id     INTEGER NOT NULL REFERENCES productos(id_producto) ON DELETE RESTRICT,
    cantidad        INTEGER NOT NULL CHECK (cantidad > 0),
    costo_estimado  NUMERIC(10, 2) NOT NULL CHECK (costo_estimado >= 0)
);

CREATE TABLE devoluciones_proveedor (
    id            BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    fecha         DATE NOT NULL DEFAULT CURRENT_DATE,
    proveedor_id  INTEGER NOT NULL REFERENCES proveedores(id_proveedor) ON DELETE RESTRICT,
    producto_id   INTEGER NOT NULL REFERENCES productos(id_producto) ON DELETE RESTRICT,
    lote_id       BIGINT NULL REFERENCES lotes(id) ON DELETE SET NULL,
    codigo_lote   VARCHAR(50) NOT NULL,
    cantidad      INTEGER NOT NULL CHECK (cantidad > 0),
    motivo        VARCHAR(60) NOT NULL,
    detalle       TEXT NOT NULL DEFAULT '',
    estado        VARCHAR(20) NOT NULL DEFAULT 'Pendiente'
                  CHECK (estado IN ('Pendiente', 'Reembolsada', 'Cambiada', 'Rechazada'))
);

-- ----------------------------------------------------------------------------
-- 4. MÓDULO CLIENTES (PORTAL), ENCARGOS Y RESERVAS
-- ----------------------------------------------------------------------------

CREATE TABLE clientes (
    id               BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    telefono         VARCHAR(9) NOT NULL UNIQUE,
    nombre           VARCHAR(120) NOT NULL,
    password         VARCHAR(128) NOT NULL,
    activo           BOOLEAN NOT NULL DEFAULT TRUE,
    failed_attempts  INTEGER NOT NULL DEFAULT 0 CHECK (failed_attempts >= 0),
    locked_until     TIMESTAMPTZ NULL,
    creado_en        TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE tokens_cliente (
    key         VARCHAR(40) PRIMARY KEY,
    cliente_id  BIGINT NOT NULL UNIQUE REFERENCES clientes(id) ON DELETE CASCADE,
    created     TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE encargos (
    id                BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    cliente_id        BIGINT NULL REFERENCES clientes(id) ON DELETE SET NULL,
    cliente_nombre    VARCHAR(120) NOT NULL,
    cliente_telefono  VARCHAR(30) NOT NULL,
    fecha_entrega     DATE NOT NULL,
    hora_entrega      VARCHAR(10) NOT NULL DEFAULT '16:00',
    estado            VARCHAR(20) NOT NULL DEFAULT 'Pendiente'
                      CHECK (estado IN ('Pendiente', 'En preparación', 'Entregado', 'Cancelado')),
    notas             TEXT NOT NULL DEFAULT '',
    creado_en         TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE detalle_encargos (
    id           BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    encargo_id   BIGINT NOT NULL REFERENCES encargos(id) ON DELETE CASCADE,
    producto_id  INTEGER NOT NULL REFERENCES productos(id_producto) ON DELETE RESTRICT,
    cantidad     INTEGER NOT NULL CHECK (cantidad > 0)
);

CREATE TABLE historial_encargos (
    id          BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    encargo_id  BIGINT NOT NULL REFERENCES encargos(id) ON DELETE CASCADE,
    fecha       TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    estado      VARCHAR(20) NOT NULL,
    motivo      VARCHAR(200) NOT NULL
);

-- ----------------------------------------------------------------------------
-- 5. MÓDULO FINANZAS Y CAJA
-- ----------------------------------------------------------------------------

CREATE TABLE ventas_finanzas (
    id                BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    fecha_hora        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    monto             INTEGER NOT NULL CHECK (monto > 0),
    medio_pago        VARCHAR(20) NOT NULL
                      CHECK (medio_pago IN ('Efectivo', 'Tarjeta', 'Transferencia')),
    nota              VARCHAR(255) NOT NULL DEFAULT '',
    estado            VARCHAR(15) NOT NULL DEFAULT 'Vigente'
                      CHECK (estado IN ('Vigente', 'Anulada')),
    motivo_anulacion  VARCHAR(255) NULL,
    fecha_anulacion   TIMESTAMPTZ NULL
);

CREATE TABLE correcciones_venta (
    id        BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    venta_id  BIGINT NOT NULL REFERENCES ventas_finanzas(id) ON DELETE CASCADE,
    fecha     TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    anterior  VARCHAR(120) NOT NULL,
    nuevo     VARCHAR(120) NOT NULL
);

-- ----------------------------------------------------------------------------
-- 6. MÓDULO REPORTES, CUADRE Y RESPALDOS
-- ----------------------------------------------------------------------------

CREATE TABLE dias_revisados (
    id           BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    fecha        DATE NOT NULL UNIQUE,
    nota         VARCHAR(255) NOT NULL DEFAULT 'Revisado por el dueño',
    revisado_en  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE registros_respaldo (
    id            BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    fecha_hora    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    tipo          VARCHAR(40) NOT NULL DEFAULT 'Manual',
    destino       VARCHAR(60) NOT NULL DEFAULT 'Local',
    tamano        VARCHAR(30) NOT NULL DEFAULT '1.0 KB',
    cifrado       BOOLEAN NOT NULL DEFAULT TRUE,
    integridad    VARCHAR(30) NOT NULL DEFAULT 'Verificado',
    archivo_ruta  VARCHAR(255) NOT NULL DEFAULT ''
);

-- ----------------------------------------------------------------------------
-- 7. ÍNDICES RECOMENDADOS (RENDIMIENTO FEFO, KARDEX Y REPORTES)
-- ----------------------------------------------------------------------------

CREATE INDEX idx_productos_categoria ON productos(id_categoria);
CREATE INDEX idx_productos_activo ON productos(activo);
CREATE INDEX idx_lotes_fefo ON lotes(producto_id, fecha_vencimiento) WHERE stock_actual > 0;
CREATE INDEX idx_movimientos_fecha_tipo ON movimientos_inventario(fecha_hora DESC, tipo);
CREATE INDEX idx_movimientos_producto ON movimientos_inventario(producto_id);
CREATE INDEX idx_encargos_estado_fecha ON encargos(estado, fecha_entrega);
CREATE INDEX idx_encargos_cliente ON encargos(cliente_id);
CREATE INDEX idx_ventas_finanzas_fecha_estado ON ventas_finanzas(fecha_hora DESC, estado);

COMMIT;
```
